Animated Radio button styles

Pulse Glow

HTML
<label class="radio pulse">
  <input type="radio">
  <span></span> Select
</label>
CSS
.pulse span {
  border: 2px solid #00f7ff;
}

.pulse input:checked + span {
  background: #00f7ff;
  box-shadow: 0 0 15px #00f7ff;
  animation: pulse 0.6s;
}

@keyframes pulse {
  50% { transform: scale(1.4); }
}

Sliding Dot

HTML
<label class="radio slide">
  <input type="radio">
  <span></span>
</label>
CSS
.slide span { border: 2px solid #ff4081; }

.slide span::after {
  content: "";
  width: 10px;
  height: 10px;
  background: #ff4081;
  border-radius: 50%;
  position: absolute;
  left: -15px;
  top: 4px;
  transition: 0.4s;
}

.slide input:checked + span::after {
  left: 5px;
}
